\section{Operational Semantics}\label{sec:tech:semantics}

The auxiliary big-step reduction rules for effect operators and the
small-step operational semantics of our core language are shown in
\autoref{fig:semantics}.

\begin{figure}[ht!]
{\footnotesize
{\small\begin{align*}
\text{\textbf{Messages}}& \  & m ::= \   & \zevent{op}{\overline{c}}
\qquad
\text{\textbf{Buffers}}& \  &\beta \in \mathcal{P}(m)  &
\qquad
\text{\textbf{Traces}}& \  &\alpha ::= \  &
    \emptr ~|~ m \;{::}\; \alpha ~|~ \alpha
    \listconcat \alpha
\end{align*}
\begin{align*}
  \text{\textbf{Handler Semantics }} & \fbox{$\alpha \vDash
                                       \zevent{op}{\overline{c}}
                                       \Downarrow \beta$} &
                                                            \quad\quad\quad
                                                            \text{\textbf{Operational
                                                            Semantics
                                                            }} &
                                                                 \fbox{$\phi
                                                                 \Downarrow v
                                                                 \quad
                                                                 \steptr{\alpha}{(\buf, e)}{\alpha}{(\buf, e)}$}
\end{align*}}
\mprooftr{65mm}{
$\buf = \{\zevent{op}{\overline{c}}\} \cup \buf_1 $ \quad
$\alpha \vDash \zevent{op}{\overline{c}} \Downarrow \buf_2$ \quad
$e' = e[\overline{x \mapsto c}]$
}{StObs}
{$\steptr{\alpha}{(\buf,
    \zobs{\overline{x}}{op}{e})}{[\zevent{op}{\overline{c}}]}{(\buf_1 \cup \buf_2, e')}$}
\quad
\mprooftr{48mm}{
$\alpha \vDash \zevent{op}{\overline{c}} \Downarrow \beta'$
}{StGen}
{$\steptr{\alpha}{(\buf, \zgen{op}{\overline{c}}{e})}{[\zevent{op}{\overline{c}}]}{(\buf \cup \beta', e)}$}
\\ \ \\ \ \\
\mprooftr{55mm}{
$\primop\ \overline{c} \Downarrow c_x$
}{StOp}
{$\steptr{\alpha}{(\buf, \zlet{x}{\primop\ \overline{c}}{e})}{[]}{(\buf, e\subst{x}{c_x})}$}
\quad
\\ \ \\ \ \\
\mprooftr{35mm}{}{StChoice1}{
$\steptr{\alpha}{(\buf, e_1 \oplus e_2)}{[]}{(\buf, e_1)}$
}\quad
\mprooftr{35mm}{}{StChoice2}{
$\steptr{\alpha}{(\buf, e_1 \oplus e_2)}{[]}{(\buf, e_2)}$
}
\\ \ \\ \ \\
\mprooftr{60mm}{
$\phi\msubst{x}{c}$
}{StAssume}{
$\steptr{\alpha}{(\buf, \zassume{\phi}{e})}{[]}{(\buf, e\msubst{x}{c})}$
}
\quad
\mprooftr{40mm}{}{StAssert}{
$\steptr{\alpha}{(\buf, \zassert{\true}{e}}{[]}{(\buf, e)}$
}
}
