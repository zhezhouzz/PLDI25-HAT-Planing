\section{Basic Typing Rules}\label{sec:tech:basic-typing}

The basic typing rules of our core language and qualifiers are shown
in \autoref{fig:basic-type-rules} and
\autoref{fig:basic-qualifier-type-rules}. We use an auxiliary function
$\Code{Ty}$ to provide a basic type for the primitives of our language,
e.g., constants, built-in operators, and data constructors.

\begin{figure}[h!]
 {\footnotesize
 {\small
\begin{flalign*}
 &\text{\textbf{Basic Typing }} & \fbox{$\Gamma \basicvdash e : s$}
\end{flalign*}}
\\ \
\begin{prooftree}
\hypo{}
\infer1[\textsc{BtConst}]{
\Gamma \basicvdash c : \Code{Ty}(c)
}
\end{prooftree}
\quad
\begin{prooftree}
\hypo{ \Gamma(x) = s }
\infer1[\textsc{BtVar}]{
\Gamma \basicvdash x : s
}
\end{prooftree}
\quad
\begin{prooftree}
\hypo{ \Gamma, x{:}b \basicvdash e : s }
\infer1[\textsc{BtFun}]{
\Gamma \basicvdash \zlam{x}{b}{e} : b\sarr s
}
\end{prooftree}
\quad
\begin{prooftree}
  \hypo{ \Gamma, f{:}b\sarr s, x{:}b \basicvdash e : s }
  \infer1[\textsc{BtFix}]{
  \Gamma \basicvdash \zfix{f}{b\sarr s}{x}{b}{e} : b\sarr s
  }
\end{prooftree}
\\[0.8em]
\begin{prooftree}
  \hypo{\Code{Ty}(\eff{op}) = \overline{s_i}{\sarr}s_x \quad \Gamma, \overline{x_i{:}s_x} \basicvdash e : \Unit}
  \infer1[\textsc{BtObs}]{
  \Gamma \basicvdash \zobs{\overline{x_i}}{op}{e} : \Unit
  }
  \end{prooftree}
\quad
\begin{prooftree}
\hypo{\Code{Ty}(\eff{op}) = \overline{s_i}{\sarr}s_x \quad \forall i. \Gamma \basicvdash v_i : s_i \quad \Gamma \basicvdash e : \Unit}
\infer1[\textsc{BtGen}]{
\Gamma \basicvdash \zgen{op}{\overline{v}}{e} : \Unit
}
\end{prooftree}
\\[0.8em]
\begin{prooftree}
  \hypo{\Code{Ty}(\primop) = \overline{s_i}{\sarr}s_x \quad \forall i. \Gamma \basicvdash v_i : s_i \quad \Gamma, x{:}s_x \basicvdash e : \Unit}
  \infer1[\textsc{BtPureOp}]{
  \Gamma \basicvdash \zlet{x}{\primop\, \overline{v_i}}{e} : \Unit
  }
\end{prooftree}
\quad
\begin{prooftree}
  \hypo{\Gamma \basicvdash e_1 : s
  \quad \Gamma \basicvdash e_2 : s}
  \infer1[\textsc{BtChoice}]{
  \Gamma \basicvdash e_1 \oplus e_2 : s
  }
\end{prooftree}
\\[0.8em]
\begin{prooftree}
\hypo{\Gamma, \overline{x{:}b} \basicvdash \phi : \Code{bool} \quad \Gamma, \overline{x{:}b} \basicvdash e : \Unit}
\infer1[\textsc{BtAssume}]{
\Gamma \basicvdash \zassume{\overline{x{:}b}}{\phi}{e} : \Unit
}
\end{prooftree}
\quad
\begin{prooftree}
\hypo{\Gamma \basicvdash \phi : \Code{bool} \quad \Gamma \basicvdash e : \Unit}
\infer1[\textsc{BtAssert}]{
\Gamma \basicvdash \zassert{\phi}{e} : \Unit
}
\end{prooftree}
\\[0.8em]
\begin{prooftree}
  \hypo{\Gamma\basicvdash e : \Unit \quad \Gamma \basicvdash e' : \Unit}
  \infer1[\textsc{BtLetE}]{
  \Gamma \basicvdash \zlet{x{:}\Unit}{e}{e'} : \Unit
  }
\end{prooftree}
  \quad
\begin{prooftree}
  \hypo{\Gamma \basicvdash v_1 : s\sarr s' \quad \Gamma \basicvdash v_2 : s \quad \Gamma, x{:}s' \basicvdash e : \Unit}
  \infer1[\textsc{BtApp}]{
  \Gamma \basicvdash \zlet{x{:}s}{v_1\;v_2}{e} : \Unit
  }
\end{prooftree}
}
\caption{Basic Typing Rules}
    \label{fig:basic-type-rules}
\end{figure}

\begin{figure}[th!]
 {\footnotesize
 {\small
\begin{flalign*}
 &\text{\textbf{Basic Qualifier Typing }} & \fbox{$\Gamma \basicvdash \phi: s$}
\end{flalign*}}
\\ \
\begin{prooftree}
\hypo{\Code{Ty}(c) = s }
\infer1[\textsc{BtLitConst}]{
\Gamma \basicvdash c : s
}
\end{prooftree}
\quad
\begin{prooftree}
\hypo{ \Gamma(x) = s }
\infer1[\textsc{BtLitVar}]{
\Gamma \basicvdash x : s
}
\end{prooftree}
\quad
\begin{prooftree}
\hypo{ }
\infer1[\textsc{BtTop}]{
\Gamma \basicvdash \top : \Code{bool}
}
\end{prooftree}
\quad
\begin{prooftree}
\hypo{ }
\infer1[\textsc{BtBot}]{
\Gamma \basicvdash \bot : \Code{bool}
}
\end{prooftree}
\\ \ \\ \ \\
\begin{prooftree}
\hypo{\Code{Ty}(\primop) = \overline{s_i}{\sarr}s \quad \forall i. \Gamma \basicvdash l_i : s_i }
\infer1[\textsc{BtLitOp}]{
\Gamma \basicvdash \primop\,\overline{l_i} : s
}
\end{prooftree}
\quad
\begin{prooftree}
\hypo{
\Gamma \basicvdash \phi : \Code{bool} }
\infer1[\textsc{BtNeg}]{
\Gamma \basicvdash \neg \phi : \Code{bool}
}
\end{prooftree}
\\ \ \\ \ \\
\begin{prooftree}
\hypo{
\Gamma \basicvdash \phi_1 : \Code{bool} \quad
\Gamma \basicvdash \phi_2 : \Code{bool} }
\infer1[\textsc{BtAnd}]{
\Gamma \basicvdash \phi_1 \land \phi_2 : \Code{bool}
}
\end{prooftree}
\quad
\begin{prooftree}
\hypo{
\Gamma \basicvdash \phi_1 : \Code{bool} \quad
\Gamma \basicvdash \phi_2 : \Code{bool} }
\infer1[\textsc{BtOr}]{
\Gamma \basicvdash \phi_1 \lor \phi_2 : \Code{bool}
}
\end{prooftree}
\quad
\begin{prooftree}
\hypo{
\Gamma, x{:}b \basicvdash \phi : \Code{bool} }
\infer1[\textsc{BtForall}]{
\Gamma \basicvdash \forall x{:}b. \phi : \Code{bool}
}
\end{prooftree}
}
