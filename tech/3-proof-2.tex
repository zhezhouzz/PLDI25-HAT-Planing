\subsection{Realizability} 

A trace realizes buffer $\{\overline{m_i}\}$ when it contains all messages in this buffer, i.e., 
{\small$\alpha_1 \listconcat [m_1] \listconcat ... [m_{n}] \listconcat \alpha_{n+1}$}. We generate this idea to automata.

\begin{definition}\label{def:beta-B}[Automata Realize Capability] A automata $F$ realizes the capability, denoted as $(\Theta, \{\overline{m_i}\}) \sim F$, when
 {\small\begin{align*}
 \exists \alpha \in \denot{F} \land \exists \overline{\alpha_i}. \alpha_1 \listconcat [m_1] \listconcat ... [m_{n}] \listconcat \alpha_{n+1} = \alpha \land 
\forall \eff{op}. \eff{op} \in \Theta \iff (\exists \overline{c}. \zevent{op}{\overline{c}} \in \{\overline{m_i}\})
\end{align*}}
\end{definition}

Then, the prophecy automata in our type judgment always contains at least one trace that realizes the capability.

\begin{lemma}\label{lemma:cap-prophecy}[Prophecy automata realize capability]
 {\small\begin{align*} 
 \Gamma;\Delta\;\Theta \vdash e : \rg{H}{A}{F} \impl \forall \sigma \in\denot{\Gamma}. \exists \beta. (\Theta, \beta) \sim \sigma(F)
\end{align*}}
\end{lemma}

We also need a lemma about buffer which can be proved by basic properties of multi-set.

\begin{lemma}\label{lemma:buf-partition}[Buffer Partition] For all capability $\Theta$, automata $F$ and buffer $\beta$, we have
\begin{align*}
    (\Theta_1 \cup \Theta_2, \beta) \sim F \impl
    \exists \beta_1\;\beta_2. \beta_1 \cup \beta_2 = \beta \land \beta_1 \cap \beta_2 = \emptyset \land (\Theta_1, \beta_1) \sim F \land (\Theta_2, \beta_2) \sim F
\end{align*}
\end{lemma}


\begin{theorem}\label{theorem:realizable}[Realizability] Given a well-formed handler specification $\Delta$, A well typed program $e$ at least realize one trace:
 {\small\begin{align*} 
 \Gamma;\Delta\;\Theta \vdash e : \rg{H}{A}{F} \impl 
 \forall \sigma \in \denot{\Gamma}.\forall \alpha_h \in \denot{\sigma(H)}. \forall \beta. (\Theta, \beta) \sim \sigma(F) \impl \exists \alpha \in\denot{\sigma(A)}. \msteptr{\alpha_h}{(\beta, e)}{\alpha}{(\emptyset, ())}
\end{align*}}
\end{theorem}
\begin{proof} We proceed by induction over our type judgment $\Gamma;\Delta;\Theta \vdash e : \tau$, which has $8$ cases proved as following:
\begin{enumerate}[label=Case]

\item:
{\footnotesize
\mprooftr{87mm}
{
$\Delta(\eff{op}) =\langle \mykw{gen}\;\tau, \Theta' \rangle$ \quad
$\Gamma \vdash \tau <: \overline{x_i{:}t_i}\sarr
\rg{H}{\lastA\msg{op}{\phi}}{A \seqA F}$ \quad
$\forall i. \Gamma \vdash v_i : t_i$ \quad
 $\Gamma; \Delta; \Theta \cup \Theta' \vdash e : \rg{H \seqA \lastA\msg{op}{\phi\msubst{x_i}{v_i}}}{A}{F}$
 }
{TGen}
{$\Gamma; \Delta; \Theta \vdash \zgen{op}{\overline{v_i}}{e} : \rg{H}{\lastA\msg{op}{\phi\msubst{x_i}{v_i}} \seqA A}{F}$}
\ \\ \ \\}
This rule assume that $e \equiv \zgen{op}{\overline{v}}{e}, \tau \equiv \rg{H}{\lastA\msg{op}{\phi\msubst{x}{v}} \seqA A}{F}$, thus we need to prove
{\footnotesize\begin{align*}
     \forall \sigma \in \denot{\Gamma}.\forall \alpha_h \in \denot{\sigma(H)}. \exists \beta. \Theta \vdash \beta \land \exists \alpha \in\denot{\sigma(\lastA\msg{op}{\phi\msubst{x_i}{v_i}} \seqA A)}. \msteptr{\alpha_h}{(\beta, \zgen{op}{\overline{v}}{e})}{\alpha}{(\emptyset, ())}
\end{align*}}\noindent
From the induction hypothesis and the precondition of this rule, we have
{\footnotesize\begin{alignat}{2}
\setcounter{equation}{0}
&\Delta(\eff{op}) =\langle \mykw{gen}\;\tau, \Theta' \rangle \ptag{assumption} \label{real:1-1} \\
&\Gamma \vdash \tau <: \overline{x{:}t}\sarr \rg{H}{\lastA\msg{op}{\phi}}{A \seqA F} \ptag{assumption} \label{real:1-2} \\
&\forall i. \Gamma \vdash v_i : t_i \ptag{assumption} \label{real:1-3} \\
&\Gamma ~|~ \Theta \cup \Theta' \vdash e : \rg{H \seqA \lastA\msg{op}{\phi\msubst{x}{v}}}{A}{F} \ptag{assumption} \label{real:1-4} \\
&\forall \sigma \in \denot{\Gamma}.\forall \alpha_h \in \denot{\sigma(H \seqA \lastA\msg{op}{\phi\msubst{x}{v}})}. \notag
\\&\quad \forall \beta. (\Theta \cup \Theta', \beta) \sim \sigma(F) \land \exists \alpha \in\denot{\sigma(A)}. \msteptr{\alpha_h}{(\beta, e)}{\alpha}{(\emptyset, ())}
\ptag{induction hypothesis} \label{real:1-5} \\
&\forall i. \forall \sigma \in \denot{\Gamma}. \sigma(v_i) \in  \sigma(\denot{t_i}) \ptag{\ref{real:1-3} and Lemma~\ref{theorem:pure-fundamental-ap}} \label{real:1-6} \\
&\forall \sigma \in \denot{\Gamma}. \sigma(\phi)\msubst{x_i}{v_i} \ptag{Lemma~\ref{lemma:built-in-typing-sub}, \ref{real:1-1}, \ref{real:1-2}, and \ref{real:1-3}} \label{real:1-7}
\\&\forall \sigma \in\denot{\Gamma}. \exists \beta. (\Theta \cup \Theta', \beta) \sim \sigma(F) \ptag{Lemma~\ref{lemma:cap-prophecy} and \ref{real:1-4}} \label{real:1-8}
\end{alignat}}\noindent
From now, we consider each $\sigma \in \denot{\Gamma}$ and $\alpha_h \in \denot{H}$ and try to prove the subgoal of this case: 
{\footnotesize\begin{align*}
    \exists \beta. \Theta \vdash \beta \land \exists \alpha \in\denot{\sigma(\lastA\msg{op}{\phi\msubst{x_i}{v_i}} \seqA A)}. \msteptr{\alpha_h}{(\beta, \zgen{op}{\overline{v}}{e})}{\alpha}{(\emptyset, ())}
\end{align*}}\noindent
Then we have
{\footnotesize\begin{alignat}{2}
&\sigma \in \denot{\Gamma} \land \alpha_h \in \denot{\sigma(H)} \ptag{assumption} \label{real:1-9}
\\& [\zevent{op}{\overline{\sigma(v_i)}}] \in \denot{\lastA\msg{op}{\sigma(\phi)\msubst{x_i}{v_i}}} \ptag{lemma~\ref{lemma:lastA-denot} and \ref{real:1-8}} \label{real:1-10}
\\& \alpha_h \listconcat [\zevent{op}{\overline{\sigma(v_i)}}] \in \denot{\sigma(H \seqA \lastA\msg{op}{\phi\msubst{x_i}{v_i}})} \ptag{lemma~\ref{lemma:seqA-denot}, \ref{real:1-9}, and \ref{real:1-10}} \label{real:1-11}
\end{alignat}}\noindent
Now, we can apply hypothesis~\ref{real:1-5} with 
{\footnotesize\begin{align*}
    &\sigma \mapsto \sigma \quad \alpha_h \mapsto \alpha_h \listconcat [\zevent{op}{\overline{\sigma(v_i)}}]
\end{align*}}\noindent
Then we have
{\footnotesize\begin{alignat}{2}
&\exists \alpha \in\denot{\sigma(A)}. \msteptr{\alpha_h}{(\beta, e)}{\alpha}{(\emptyset, ())} \ptag{hypothesis~\ref{real:1-5} with \ref{real:1-8}, \ref{real:1-9}, \ref{real:1-11}} \label{real:1-12}
\\& \exists \beta_1\;\beta_2. \beta_1 \cup \beta_2 = \beta \land \beta_1 \cap \beta_2 = \emptyset \land (\Theta, \beta_1) \sim F \land (\Theta', \beta_2) \sim F \ptag{Lemma~\ref{lemma:buf-partition}} \label{real:1-13}
\\& \alpha \listconcat [\zevent{op}{\overline{\sigma(v_i)}}] \in \denot{\sigma(\lastA\msg{op}{\phi\msubst{x_i}{v_i}} \seqA A)} \ptag{lemma~\ref{lemma:seqA-denot}, \ref{real:1-10}, and \ref{real:1-12}} \label{real:1-14}
\\&\exists \alpha_f \in \denot{F} \land \exists \overline{\alpha_i}. \exists \overline{m_i}.\; \alpha_1 \listconcat [m_1] \listconcat ... [m_{n}] \listconcat \alpha_{n+1} = \alpha_f \land \{\overline{m_i}\} = \beta \ptag{definition~\ref{def:beta-B} and \ref{real:1-8}} \label{real:1-15}
\end{alignat}}\noindent
With help of hypothesis \ref{real:1-13} and \ref{real:1-14}, we can instantiate the existential quantified variables as $\beta \mapsto \beta_1, \alpha \mapsto [\zevent{op}{\overline{\sigma(v_i)}}] \listconcat \alpha$, and we need to prove
{\footnotesize\begin{align*}
    \msteptr{\alpha_h}{(\beta_1, \zgen{op}{\overline{v}}{e})}{ \alpha \mapsto [\zevent{op}{\overline{\sigma(v_i)}}] \listconcat \alpha}{(\emptyset, ())}
\end{align*}}\noindent
According to the well-formed type context (Lemma~\ref{lemma:built-in-typing-sub}), \ref{real:1-1}, \ref{real:1-2}, \ref{real:1-7}, and \ref{real:1-15}, we have
{\footnotesize\begin{alignat}{2}
& \alpha_h \vDash \zevent{op}{\overline{c}} \Downarrow \beta_2 \ptag{Lemma~\ref{lemma:built-in-typing-sub}} \label{real:1-16}
\\& \msteptr{\alpha_h}{(\beta_1 \cup \beta_2, e)}{\alpha}{(\emptyset, ())} \ptag{ \ref{real:1-12} and \ref{real:1-13}} \label{real:1-17}
\\& \msteptr{\alpha_h}{(\beta_1, \zgen{op}{\overline{v}}{e})}{ \alpha \mapsto [\zevent{op}{\overline{\sigma(v_i)}}] \listconcat \alpha}{(\emptyset, ())} \ptag{\textsc{StGen}, \ref{real:1-17}, and \ref{real:1-12}} \label{real:1-18}
\end{alignat}}\noindent
which is sufficient to prove the subgoal in this case.

From now, we consider each $\sigma\msubst{x_i}{v_i} \in \denot{\Gamma, \overline{x{:}t}}$ and $\alpha_h \in \denot{H}$ and try to prove the subgoal of this case:

& \forall \sigma \in \denot{\Gamma,  x{:}t_x\msubst{y}{v}}.\forall \alpha_h \in \denot{\sigma(H)}. \forall \beta. (\Theta, \beta) \sim \sigma(F) \impl 
\\&\quad \exists \alpha \in\denot{\sigma(A)}. \msteptr{\alpha_h}{(\beta, e)}{\alpha}{(\emptyset, ())}
\ptag{induction hypothesis and \ref{real:3-4}} \label{real:3-6}
\end{alignat}}\noindent
This reduction step is pure, thus we can directly instantiate $\alpha$ in subgoal as $\alpha$ and apply hypothesis~\ref{real:3-6}, then which is sufficient to prove this case.

\item:
{\footnotesize
\mprooftr{38mm}
{$\Gamma;\Delta;\Theta \vdash e : \tau$ \quad
$\Gamma \vdash \tau <: \tau'$}
{TSub}
{$\Gamma;\Delta;\Theta \vdash e : \tau'$}
\ \\ \ \\}
The case can be directly proved by Lemma~\ref{lemma:subtyping}.

\end{enumerate}
\end{proof}

\subsection{Type Soundness} 

The type soundness can be proved by fundamental theorem and realizability.

\begin{theorem}\label{theorem:sound-ap}[Type Soundness] Given
  a well-formed handler specification $\Delta$, with ghost variables $\overline{x{:}b}$ and a violation property $A$, a controller $e$ that satisfies
  $\overline{x{:\rawnuot{b}{\top}}};\Delta; \emptyset \vdash e : \rg{\epsTL}{A}{\epsTL}$, then $e$ at least realize one trace consistent with $A$:
 {\footnotesize\begin{align*} \exists
    \overline{c{:}b}. \exists \alpha. \msteptr{[]}{(\emptyset,
      e[\overline{x\mapsto c}])}{\alpha}{(\emptyset, ())} \land \alpha \in
    \denot{A[\overline{x\mapsto c}]}
\end{align*}}
\end{theorem}
\begin{proof} According to the fundamental theorem, we have
{\footnotesize\begin{alignat}{2}
\setcounter{equation}{0}
    &\overline{x{:\rawnuot{b}{\top}}} ;\Delta; \emptyset \vdash e : \rg{\epsTL}{A}{\epsTL} \ptag{assumption} \label{sound:1}
    \\&\forall \sigma, \sigma \in
    \denot{\overline{x{:\rawnuot{b}{\top}}}} \impl \sigma(e) \in
    \denot{\sigma(\rg{\epsTL}{A}{\epsTL})} \ptag{\autoref{theorem:fundamental-ap} and \ref{sound:1}} \label{sound:2}
    \\&\forall \sigma, \sigma \in
    \denot{\overline{x{:\rawnuot{b}{\top}}}} \iff \exists \overline{c{:}b}. \sigma =\msubst{x}{c}  \ptag{definition of $\denot{\Gamma}$ and \ref{sound:2}}  \label{sound:3}
    \\&\forall \overline{c{:}b}. e\msubst{x}{c} \in
    \denot{\rg{\epsTL}{A\msubst{x}{c}}{\epsTL}} \ptag{hypothesis~\ref{sound:2} and \ref{sound:3}} \label{sound:4}
    \\& \forall \sigma \in \denot{\overline{x{:\rawnuot{b}{\top}}}}.\forall \alpha_h \in \denot{\sigma(\epsTL)}. \forall \beta. (\emptyset, \beta) \sim \sigma(\epsTL) \impl \notag \\&\quad\exists \alpha \in\denot{\sigma(A)}. \msteptr{\alpha_h}{(\beta, e)}{\alpha}{(\emptyset, ())} 
    \ptag{\autoref{theorem:realizable} and \ref{sound:1}} \label{sound:5}
    \\&\exists \overline{c{:}b}. \exists \alpha. \msteptr{[]}{(\emptyset, e\msubst{x}{c})}{\alpha}{(\emptyset, ())} 
    \ptag{Lemma~\ref{lemma:espTL} and \ref{sound:5}} \label{sound:6}
\end{alignat}}\noindent
the definition of \PAT denotation as shown in \autoref{fig:type-denotation} indicates
{\footnotesize\begin{align*}
    &\forall \alpha_h \in\denot{\sigma(\epsTL)}.\forall \alpha_f \in\denot{\sigma(\epsTL)}. \forall \alpha\; \beta\;\beta'\;e_h\;e_f.
  \\&\qquad \msteptr{[]}{(\emptyset, e_h)}{\alpha_h}{(\beta, ())}\land \msteptr{\alpha_h}{(\beta, e)}{\alpha}{(\beta', ())} \land \msteptr{\alpha_h \listconcat \alpha }{(\beta', e_f)}{\alpha_f}{(\emptyset, ())} \impl \alpha \in\denot{A\msubst{x}{c}}
\end{align*}}\noindent
Again, according to Lemma~\ref{lemma:espTL}, we have
{\footnotesize\begin{alignat}{2}
  &\forall \alpha\;\beta\;\beta'\;e_h\;e_f. \notag
  \\&\quad \msteptr{[]}{(\emptyset, e_h)}{[]}{(\beta, ())} \land \msteptr{[]}{(\beta, e)}{\alpha}{(\beta', ())} \land \msteptr{\alpha }{(\beta', e_f)}{[]}{(\emptyset, ())} \impl \alpha \in\denot{A} \label{sound:7}
  \\&\msteptr{[]}{(\emptyset, ())}{[]}{(\emptyset, ())} \ptag{definition of $\hookrightarrow^*$} \label{sound:8}
  \\&\msteptr{\alpha}{(\emptyset, ())}{[]}{(\emptyset, ())} \ptag{definition of $\hookrightarrow^*$} \label{sound:9}
  \\&\exists \overline{c{:}b}.\exists \alpha \in\denot{\sigma(A\msubst{x}{c})}. \msteptr{[]}{(\emptyset, e)}{\alpha}{(\emptyset, ())}
\ptag{hypothesis~\ref{sound:7}, \ref{sound:6},\ref{sound:8}, and \ref{sound:9}}
  \label{sound:10}
\end{alignat}}\noindent
This is sufficient to establish the original theorem we aim to prove.
\end{proof}
