\section{Auxiliary functions in Synthesis Algorithm}\label{sec:tech:algo}

The normalization function that convert an automata into a union of abstract trace is provided as auxiliary function $\normPlan$ in in \autoref{algo:norm}. The function $\termDerive$ is also provided in \autoref{algo:term-derive}.

\paragraph{Normalization} The function $\normPlan$ first convert the input automata in s-LTL$_f$ into standard Negation Normal Form (NNF) on line $1$, then inductively translate the input automata into a set of abstract traces. Notice, the nagation operator $\neg$ only appear before the atomic predicates (i.e., $\msg{op}{\phi}$ and $\evparenth{\phi}$).

\begin{algorithm}[t!]
    \Procedure{$\normPlan(A) := $}{
    $A \leftarrow \Code{ToNNF}(A)$\;
    \Match{$A$}{
      \lCase{$\msg{op}{\phi}$}{ \Return{$\{\lastA\msg{op}{\phi} \seqA \allTL \}$} }
      \lCase{$\neg \msg{op}{\phi}$}{ \Return{$\{\lastA\msg{op}{\neg \phi} \seqA \allTL \} \cup \{\lastA\msg{op'}{\top} \seqA \allTL ~|~ \eff{op'} \not= \eff{op} \}$} }
      \lCase{$\evparenth{\phi}$}{ 
      \Return{$\{ \lastA \msg{op}{\phi} ~|~ \text{ for all } \eff{op} \}$}}
      \lCase{$\neg \evparenth{\phi}$}{ 
      \Return{$\{ \lastA \msg{op}{\neg\phi} ~|~ \text{ for all } \eff{op} \}$}}
      \lCase{$\lastA A$}{ \Return{$\{\lastA\msg{op}{\phi} ~|~ \lastA\msg{op}{\phi} \seqA \Pi \in \normPlan(A) \}$} }
      \lCase{$\nextA A$}{ \Return{$\{ \msg{op}{\top} \seqA \Pi ~|~ \text{ for all } \eff{op}, \Pi \in \normPlan(A) \}$} }
      \lCase{$A_1 \untilA A_2$}{ 
      \Return{$\{ (\globalA\neg A_1) \seqA \Pi_2 ~|~ \Pi_2 \in \normPlan(A_2) \}$}}
      \lCase{$A_1 \seqA A_2$}{ \Return{$\{ \Pi_1 \seqA \Pi_2 ~|~ \Pi_1 \in \normPlan(A_1) \land \Pi_2 \in \normPlan(A_2) \}$}}
      \lCase{$\finalA A$}{ 
      \Return{$\{ \globalA \evparenth{\top} \seqA \Pi \seqA \globalA \evparenth{\top} ~|~ \Pi \in \normPlan(A) \}$}}
      \lCase{$\globalA A$}{ 
      \Return{$\{ \globalA A \}$}}
      \lCase{$A_1 \lorA A_2$}{ \Return{$\normPlan(A_1) \cup \normPlan(A_2)$}}
      \lCase{$A_1 \land A_2$}{ 
      \Return{$\{ \Pi_1 \land \Pi_2 ~|~ \Pi_1 \in \normPlan(A_1) \land \Pi_2 \in \normPlan(A_2) \}$}}
    }
    }
    \caption{Abstract Trace Normalization}
    \label{algo:norm}
  \end{algorithm}

\begin{lemma}\label{lemma:traceAnd}[Conjunction perverse abstract traces] The conjunction ($\land$) of two abstract traces is still an abstract traces.
\end{lemma}

\begin{lemma}\label{lemma:norm-sound}[Normalization Soundness] The normalized result has the same denotation as the input automata, that is, for all automata $A$ and set of traces $\{\Pi_i\}$,
{\small\begin{align*}
    \denot{A} = \bigcup_{i} \denot{\Pi_i}
\end{align*}}
\end{lemma}

\begin{algorithm}[t!]
    \Procedure{$\S{TermDerive}(\Gamma, \Pi) := $}{
      \Match{$\Gamma$}{
      \Case{$[]$}{ \Return{$\S{DeriveTrace}(\Pi)$}\; }
      \Case{$x{:}\rawnuot{b}{\phi} \cons \Gamma'$}{ 
      \Return{$\zassume{\phi[\vnu \mapsto x]}{\S{TermDerive}(\Gamma', \Code{ToList}(\Pi))}$}\;
      }
    }
    }
    \caption{Term Derivation}
    \label{algo:term-derive}
  \end{algorithm}

\begin{algorithm}[t!]
    \Procedure{$\S{DeriveTrace}(\Pi) := $}{
    \Match{$\Pi$}{
      \lCase{$[]$}{ \Return{$()$} }
      \lCase{$\globalA A \cons \Pi'$}{ \Return{$\S{DeriveTrace}(\Pi')$} }
      \Case{$\lastA \msgB{op}{\overline{x}}{\phi} \seqA \Pi'$ when $\mykw{gen}\;\eff{op}$}{
        $\overline{x'} \leftarrow \Code{GetFreshNames}(\overline{x})$\;
        $\zassume{\phi\msubst{x}{x'}}{\zgen{op}{\overline{x'}}{\S{DeriveTrace}(\Pi')}}$\;
      }
      \Case{$\lastA \msgB{op}{\overline{x}}{\phi} \cons \Pi'$ when $\mykw{obs}\;\eff{op}$}{
        $\overline{x'} \leftarrow \Code{GetFreshNames}(\overline{x})$\;
        $\zobs{\overline{x'}}{op}{\zassert{\phi\msubst{x}{x'}}{\S{DeriveTrace}(\Pi')}}$\;
      }
    }
    }
    \caption{Trace Derivation}
    \label{algo:trace-derive}
\end{algorithm}

\paragraph{Term Derivation} The term derivation function \(\S{TermDerive}\) is shown in \autoref{algo:term-derive}. It first converts the input type context into \(\mykw{assume}\) statements over the corresponding qualifiers in pure refinement types (line 6), then derives the abstract trace with the help of the function \(\S{DeriveTrace}\) shown in \autoref{algo:trace-derive}. This function requires the input abstract trace to first be converted into a list of automata (\(\Code{ToList}\)), then transforms this list into a controller program inductively. Note that our algorithm prioritizes shorter controller programs, so \(\S{DeriveTrace}\) skips automata with global modality (\(\globalA A\)) on line 4. For a symbolic event with a generative operator (line 5), \(\S{DeriveTrace}\) derives an \(\mykw{assume}\) expression before the \(\mykw{gen}\) expression on line 7. Conversely, for observable events, \(\S{DeriveTrace}\) derives an \(\mykw{assert}\) expression after the \(\mykw{obs}\) expression on line 10.

\begin{lemma}\label{lemma:derive-sound}[Term Derivation Soundness] For given type context $\Gamma$, well-founded type context $\Delta$, abstract trace $\Pi$, and term $e$,
{\small\begin{align*}
   (\exists e'. \Gamma;\Delta;\emptyset \vdash e' : \rg{\epsTL}{A}{\epsTL}) \impl
   \S{TermDerive}(\Gamma, \Pi) = e \impl \Gamma;\Delta;\emptyset \vdash e : \rg{\epsTL}{A}{\epsTL}
\end{align*}}
\end{lemma}
